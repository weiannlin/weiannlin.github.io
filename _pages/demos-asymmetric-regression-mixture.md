<div class="jd-prose" markdown="1">

Linearity of the direct [regression function](/lecture-notes/double-expectation-theorem/#thm-regression-function) $\mathbb{E}(Y\mid X=x)$ does not imply linearity of the reverse regression function $\mathbb{E}(X\mid Y=y)$. When one variable follows a symmetric two-component mixture (bimodal when $\mu>s$) and the other is generated by adding homoscedastic normal noise around a linear trend, one conditional direction stays Gaussian and strictly linear whereas the opposite conditional direction shifts probability weight between the two mixture mounds according to [Bayes’ theorem](/lecture-notes/bayes-rule-posterior-probability/#theorem-bayes-rule).

Specifically, in the **Mixture on X** configuration, let $X$ follow the symmetric two-component Gaussian mixture $f_{\sssig X}(x)=\frac{1}{2}\phi(x;-\mu,s)+\frac{1}{2}\phi(x;+\mu,s)$ ($\mu\geqslant 0$, $s>0$) and let $Y\mid(X=x)\sim\mathcal{N}(\beta x,\tau^{2})$ ($\tau>0$, $\beta\in\mathbb{R}$). Writing $v=\beta^{2}s^{2}+\tau^{2}$ for the within-component variance of $Y$, the [joint pdf](/lecture-notes/joint-probability-density-functions/#def-joint-pdf) and [marginal pdf](/lecture-notes/marginal-probability-density-functions/#def-marginal-pdf) of $Y$ take the closed forms

$$
\begin{gathered}
f_{\sssig XY}(x,y)=\frac{1}{2}\bigl[\phi(x;-\mu,s)+\phi(x;\mu,s)\bigr]\,\phi(y;\beta x,\tau)\\[0.32em]
f_{\sssig Y}(y)=\frac{1}{2}\phi\!\left(y;-\beta\mu,\sqrt{v}\right)+\frac{1}{2}\phi\!\left(y;\beta\mu,\sqrt{v}\right)
\end{gathered}
$$

Since the conditional law $Y\mid(X=x)\sim\mathcal{N}(\beta x,\tau^{2})$ does not depend on the marginal shape of $X$, the direct regression $\mathbb{E}(Y\mid X=x)=\beta x$ is an exact straight line. Slicing horizontally at $Y=y$, however, updates the posterior weights of the two components centered at $\pm\mu$ in proportion to $\exp(\pm\beta\mu y/v)$, yielding closed-form expressions for the reverse [conditional expectation](/lecture-notes/conditional-expectation-and-variance/#def-conditional-expectation) and [conditional variance](/lecture-notes/conditional-expectation-and-variance/#def-conditional-variance):

$$
\begin{gathered}
\mathbb{E}(X\mid Y=y)=\frac{\beta s^{2}}{v}\,y+\frac{\tau^{2}\mu}{v}\tanh\!\left(\frac{\beta\mu}{v}\,y\right)\\[0.32em]
\mathrm{Var}(X\mid Y=y)=\frac{s^{2}\tau^{2}}{v}+\left(\frac{\tau^{2}\mu}{v}\right)^{2}\operatorname{sech}^{2}\!\left(\frac{\beta\mu}{v}\,y\right)
\end{gathered}
$$

Toggling **Mixture on Y** exchanges the roles of $(X,Y)$, giving a curved $\tanh$ direct regression $\mathbb{E}(Y\mid X=x)$ paired with an exact straight reverse regression $\mathbb{E}(X\mid Y=y)=\beta y$.

**Parameters to explore below:**

- **Peak half-gap $\mu$ and component standard deviation $s$**: Increasing $\mu$ or decreasing $s$ separates the two mounds sharply, producing a bimodal slice near $0$ and steepening the central $\tanh$ transition; setting $\mu=0$ collapses the mixture back to a single Gaussian bell where both regressions become straight lines.
- **Linear slope $\beta$ and noise standard deviation $\tau$**: Smaller $\tau$ tightens the conditional slices around the linear trend, sharpening the S-shaped posterior transition between the two mixture components.

<script>
window.JD_CONFIG = (function () {
  const SQRT_2PI = Math.sqrt(2 * Math.PI);
  function norm(u, m, s) {
    const z = (u - m) / s;
    return Math.exp(-0.5 * z * z) / (SQRT_2PI * s);
  }
  function mix2(u, mu, s) {
    return 0.5 * norm(u, -mu, s) + 0.5 * norm(u, mu, s);
  }
  function bayesMoments(v, mu, s, beta, tau) {
    const vv = beta * beta * s * s + tau * tau;
    const arg = (beta * mu * v) / vv;
    const th = Math.tanh(arg);
    const sech2 = 1 - th * th;
    const m0 = (tau * tau * mu) / vv;
    const mean = ((beta * s * s) / vv) * v + m0 * th;
    const variance = (s * s * tau * tau) / vv + m0 * m0 * sech2;
    return { mean, sd: Math.sqrt(Math.max(1e-9, variance)) };
  }
  function bayesPosteriorPdf(u, v, mu, s, beta, tau) {
    const vv = beta * beta * s * s + tau * tau;
    const arg = Math.max(-40, Math.min(40, (beta * mu * v) / vv));
    const wPlus = 1 / (1 + Math.exp(-2 * arg));
    const wMinus = 1 - wPlus;
    const sdPost = Math.sqrt((s * s * tau * tau) / vv);
    const shift = ((beta * s * s) / vv) * v;
    const m0 = (tau * tau * mu) / vv;
    return wMinus * norm(u, shift - m0, sdPost) + wPlus * norm(u, shift + m0, sdPost);
  }

  return {
    domainL: 3.2,
    defaultMode: 'y',
    defaultSubMode: 'mix-x',
    defaultCond3D: true,
    subModes: [
      {
        id: 'mix-x',
        label: 'Mixture on X',
        title: 'Mixture on X: linear E(Y | X = x) paired with tanh E(X | Y = y)',
        onSelect: (st, setParam, setSliceMode) => { setSliceMode('y'); setParam('slice', 0.00); }
      },
      {
        id: 'mix-y',
        label: 'Mixture on Y',
        title: 'Mixture on Y: tanh E(Y | X = x) paired with linear E(X | Y = y)',
        onSelect: (st, setParam, setSliceMode) => { setSliceMode('x'); setParam('slice', 0.00); }
      }
    ],
    sliders: [
      { id: 'slice', label: 'Slice y₀', dynamicLabel: st => st.mode === 'x' ? 'Slice position x₀' : 'Slice position y₀', min: -2.50, max: 2.50, step: 0.05, def: 0.00, signed: true },
      { id: 'mu', label: 'Peak half-gap μ', min: 0.00, max: 1.75, step: 0.05, def: 1.35, signed: false },
      { id: 's', label: 'Component SD s', min: 0.30, max: 0.85, step: 0.02, def: 0.42, signed: false },
      { id: 'beta', label: 'Linear slope β', min: -1.15, max: 1.15, step: 0.05, def: 0.80, signed: true },
      { id: 'tau', label: 'Noise SD τ', min: 0.32, max: 1.00, step: 0.02, def: 0.52, signed: false }
    ],
    readoutLabels: [
      'Slice area (side-wall marginal)',
      'Renormalization factor',
      'Active slice mean & SD',
      'Regression linearity pair'
    ],
    evalJoint: (x, y, p, st) => st.subMode === 'mix-x'
      ? mix2(x, p.mu, p.s) * norm(y, p.beta * x, p.tau)
      : mix2(y, p.mu, p.s) * norm(x, p.beta * y, p.tau),
    evalMargX: (x, p, st) => st.subMode === 'mix-x'
      ? mix2(x, p.mu, p.s)
      : mix2(x, p.beta * p.mu, Math.sqrt(p.beta * p.beta * p.s * p.s + p.tau * p.tau)),
    evalMargY: (y, p, st) => st.subMode === 'mix-x'
      ? mix2(y, p.beta * p.mu, Math.sqrt(p.beta * p.beta * p.s * p.s + p.tau * p.tau))
      : mix2(y, p.mu, p.s),
    evalCond: (t, sv, mode, p, st) => {
      if (st.subMode === 'mix-x') {
        return mode === 'x' ? norm(t, p.beta * sv, p.tau) : bayesPosteriorPdf(t, sv, p.mu, p.s, p.beta, p.tau);
      }
      return mode === 'y' ? norm(t, p.beta * sv, p.tau) : bayesPosteriorPdf(t, sv, p.mu, p.s, p.beta, p.tau);
    },
    evalDirect: (x, p, st) => st.subMode === 'mix-x'
      ? { mean: p.beta * x, sd: p.tau }
      : bayesMoments(x, p.mu, p.s, p.beta, p.tau),
    evalReverse: (y, p, st) => st.subMode === 'mix-x'
      ? bayesMoments(y, p.mu, p.s, p.beta, p.tau)
      : { mean: p.beta * y, sd: p.tau },
    formatReadouts: (sc, st, fmt) => {
      const p = st.params;
      const sv = fmt(p.slice, 2, true);
      const area = sc.sliceArea;
      const mult = 1 / Math.max(1e-30, area);
      const multStr = mult >= 1000 ? `×${mult.toExponential(1)}` : `×${mult.toFixed(2)}`;
      const dirTxt = mult > 1.02 ? 'stretch up' : (mult < 0.98 ? 'scale down' : 'unit scale');
      const mSym = st.mode === 'x' ? `f_X(${sv})` : `f_Y(${sv})`;
      const cSym = st.mode === 'x' ? `E(Y | X = ${sv})` : `E(X | Y = ${sv})`;
      const pairTxt = (p.mu < 1e-4 || Math.abs(p.beta) < 1e-4)
        ? 'Both linear (μ = 0 or β = 0)'
        : (st.subMode === 'mix-x' ? 'E(Y | X = x) linear · E(X | Y = y) tanh' : 'E(Y | X = x) tanh · E(X | Y = y) linear');
      return [
        `${mSym} = ${area < 1e-3 ? area.toExponential(2) : area.toFixed(4)}`,
        `1 / ${mSym} = ${multStr} (${dirTxt})`,
        `${cSym} = ${fmt(sc.condMean, 3, true)} · SD = ${sc.condSd.toFixed(3)}`,
        pairTxt
      ];
    }
  };
})();
</script>
